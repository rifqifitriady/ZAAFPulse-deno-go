# The token saver on the Go engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port `lib/gateway/rtk.js` (the token saver) to the Go engine byte for byte, so that requests with tool results
are served by Go instead of being handed to JS, behind a `token-saver` switch with its own shadow record.

**Architecture:** New files in `gw/engine` (`jsstr.go`, `rtk.go`, `rtk_json.go`, `rtk_body.go`) hold the text layer
(UTF-16 code units, rtk.js's regex sources run on `gw/guard/jsre`, which gains the `m` flag) and the two body walks.
Planning runs the cheap structural walk (hints, throws, candidate texts); `State.Save` runs the filters before `Take`.
Proof is fixtures from the real JS (`scripts/fixtures/gen-rtk.js`, `saver-*` cases in `gen-chat-guard.js`, P4d's
`gen-xlate.js`), the shadow, and the contract suite.

**Tech Stack:** Go 1.27 (stdlib only, `gw/guard/jsre`, `gw/engine` JSON), Node 24 and Deno 2.9 for the generators,
`node --test` for the JS side.

Spec: `docs/superpowers/specs/2026-11-17-rtk-go-design.md` (Decisions 1–20, R1–R12 cited below as "spec D<n>", "R<n>").

## Global Constraints

- Branch `migrate/p4-rtk-go`; never commit `.superpowers/`; no temporary branches; no `git stash`; no AI attribution
  in commits; do not touch ports 4777 or 4787.
- Go: stdlib only; `gofmt -l gw` empty, `go -C gw vet ./...` and `GOOS=windows go -C gw vet ./...` clean,
  `go -C gw test -race ./...` green.
- Every generator under `scripts/fixtures/` must give identical bytes on Node and on Deno with
  `deno run -q --no-prompt --allow-read --allow-write=gw/testdata --deny-env --deny-net --deny-run --deny-sys`.
- No fixture file over 512 KiB; inputs over 64 KiB are recipes; outputs over 64 KiB are `{ length, sha256 }`.
- Thresholds are rtk.js's, in UTF-16 units: 600, 10 MiB (10485760), 2048, 64, 2000, 4000; `savedTokens =
  ⌊(savedChars + 2) / 4⌋`.
- Hand-off reasons (exact text): `token saver: a message it cannot read`, `token saver: a tool call the engine cannot
  read`, `token saver: a JSON tool result the engine cannot hold`, `token saver: over its step budget`.
- Step budget: `64 × units + 1<<24` steps, 2 s deadline.
- The switch is named `token-saver` everywhere (config, schema, shadow log, Go).
- Tasks 1–5 touch no P4d file and may start now. Tasks 6–10 start only after P4d is merged into `migrate/deno-go` and
  `migrate/deno-go` is merged into this branch; every step marked **(re-check after P4d merges)** names code P4d
  changes: read the merged file first and apply the step to its current shape.

// JavaScript string semantics the token saver's port needs (spec D3, D5), on UTF-16 code units: String.length,
// split('\n'), trim, the part of toLowerCase the hint patterns can see, a template literal's ToString of a JSON value,
// and Map keys under SameValueZero.

import (
	"math"
	"strconv"
	"unicode/utf16"

	"github.com/rifqifitriady/ZAAFPulse-deno-go/gw/guard/jsre"
)

func u16(s string) []uint16 { return jsre.U16(s) }

// units is String.prototype.length of a Go string.
func units(s string) int {
	n := 0
	for _, r := range s {
		n += utf16.RuneLen(r)
	}
	return n
}

func cat(parts ...[]uint16) []uint16 {
	n := 0
	for _, p := range parts {
		n += len(p)
	}
	out := make([]uint16, 0, n)
	for _, p := range parts {
		out = append(out, p...)
	}
	return out
}

// splitNL is String.prototype.split('\n'): n separators give n+1 parts.
func splitNL(t []uint16) [][]uint16 {
	out := make([][]uint16, 0, 16)
	start := 0
	for i, c := range t {
		if c == '\n' {
			out = append(out, t[start:i:i])
			start = i + 1
		}
	}
	return append(out, t[start:len(t):len(t)])
}

// eachLine calls f with each part of split('\n') in order, without building the list (spec D19).
func eachLine(t []uint16, f func(l []uint16)) {
	start := 0
	for i, c := range t {
		if c == '\n' {
			f(t[start:i:i])
			start = i + 1
		}
	}
	f(t[start:len(t):len(t)])
}

// joinNL is Array.prototype.join('\n').
func joinNL(ls [][]uint16) []uint16 {
	n := 0
	for _, l := range ls {
		n += len(l) + 1
	}
	out := make([]uint16, 0, n)
	for i, l := range ls {
		if i > 0 {
			out = append(out, '\n')
		}
		out = append(out, l...)
	}
	return out
}

// trim16 is String.prototype.trim.
func trim16(t []uint16) []uint16 {
	i, j := 0, len(t)
	for i < j && jsre.IsSpace(t[i]) {
		i++
	}
	for j > i && jsre.IsSpace(t[j-1]) {
		j--
	}
	return t[i:j]
}

func blank(t []uint16) bool { return len(trim16(t)) == 0 }

// hasPrefix16: p is ASCII.
func hasPrefix16(t []uint16, p string) bool {
	if len(t) < len(p) {
		return false
	}
	for i := 0; i < len(p); i++ {
		if t[i] != uint16(p[i]) {
			return false
		}
	}
	return true
}

func equal16(a, b []uint16) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}

func lastIndex16(t []uint16, c uint16) int {
	for i := len(t) - 1; i >= 0; i-- {
		if t[i] == c {
			return i
		}
	}
	return -1
}

func itoa16(n int) []uint16 { return u16(strconv.Itoa(n)) }

// key16 is an exact map key for a text (lone surrogates included).
func key16(t []uint16) string {
	b := make([]byte, 2*len(t))
	for i, c := range t {
		b[2*i], b[2*i+1] = byte(c>>8), byte(c)
	}
	return string(b)
}

// hintLower is String.prototype.toLowerCase as far as the hint patterns can tell (spec D5): they test ASCII letters,
// \b (ASCII word characters) and [^|;&]. Only A-Z, U+212A KELVIN SIGN (to k) and U+0130 (to i U+0307) lower-case into
// ASCII; every other mapping goes from one non-ASCII unit to another, which no hint pattern tells apart, so those are
// kept as they are.
func hintLower(t []uint16) []uint16 {
	out := make([]uint16, 0, len(t)+1)
	for _, c := range t {
		switch {
		case c >= 'A' && c <= 'Z':
			out = append(out, c+('a'-'A'))
		case c == 0x212a:
			out = append(out, 'k')
		case c == 0x130:
			out = append(out, 'i', 0x307)
		default:
			out = append(out, c)
		}
	}
	return out
}

// jsNumber is Number.prototype.toString, Infinity included (a template literal writes it; JSON writes null).
func jsNumber(n float64) string {
	switch {
	case math.IsInf(n, 1):
		return "Infinity"
	case math.IsInf(n, -1):
		return "-Infinity"
	case n == 0:
		return "0"
	}
	return NumberString(n)
}

// jsToString is a template literal's ToString of a JSON value (nil is undefined). ok is false where JavaScript throws a
// TypeError: an object with an own toString key (OrdinaryToPrimitive finds no callable method that gives a primitive),
// also inside an array (Array.prototype.join converts each item).
func jsToString(v *Value) ([]uint16, bool) {
	if v == nil {
		return u16("undefined"), true
	}
	switch v.Kind {
	case Null:
		return u16("null"), true
	case Bool:
		if v.B {
			return u16("true"), true
		}
		return u16("false"), true
	case Number:
		return u16(jsNumber(v.N)), true
	case String:
		return u16(v.S), true
	case Array:
		out := []uint16{}
		for i, x := range v.A {
			if i > 0 {
				out = append(out, ',')
			}
			if x == nil || x.Kind == Null {
				continue
			}
			s, ok := jsToString(x)
			if !ok {
				return nil, false
			}
			out = append(out, s...)
		}
		return out, true
	}
	if v.Get("toString") != nil {
		return nil, false
	}
	return u16("[object Object]"), true
}

// mapKey is a Map key under SameValueZero. ok is false for an object or an array: a parsed value is never the same
// object as another, so such a key is never found.
func mapKey(v *Value) (string, bool) {
	if v == nil {
		return "u", true
	}
	switch v.Kind {
	case Null:
		return "n", true
	case Bool:
		if v.B {
			return "t", true
		}
		return "f", true
	case Number:
		return "d" + jsNumber(v.N), true
	case String:
		return "s" + v.S, true
	}
	return "", false
}

// The token saver (lib/gateway/rtk.js) on the go engine (spec docs/superpowers/specs/2026-11-17-rtk-go-design.md): long
// tool results are shortened by a filter picked from the tool call or from the output's shape, byte for byte as the
// JS does. Text is UTF-16 code units throughout (D3); the patterns are rtk.js's own sources run on jsre (D4). A nil
// *jsre.Budget means no limit; with one, a run past it panics with a jsre limit error that Budget.Run returns.

import (
	"bytes"

	"errors"

// rtkHit is one shortened text: the filter that did it and the UTF-16 units it saved.

	Saved  int
}

func rx(src, flags string) *jsre.Regexp { return jsre.MustCompile(src, flags) }

var (
	rtkANSI       = rx(`\x1b\[[0-9;?]*[ -/]*[@-~]|\x1b\][^\x07]*(\x07|\x1b\\)`, "g")
	rtkNoiseDirs  = rx(`(^|\/)(node_modules|\.git|dist|build|\.next|\.venv|venv|__pycache__|\.cache|target|coverage)(\/|$)`, "")
	rtkHexID      = rx(`\b[0-9a-f]{7,}\b`, "gi")
	rtkDigits     = rx(`\d+`, "g")
	rtkDiffHead   = rx(`^(diff --git|index |--- |\+\+\+ )`, "")
	rtkLogHead    = rx(`^(commit |Author:|Date:|Merge:)`, "")
	rtkLogMsg     = rx(`^ {4}\S`, "")
	rtkIndent4    = rx(`^ {4}`, "")
	rtkStatusFile = rx(`^\s+(modified|new file|deleted|renamed):|^\s{1,2}[MADRCU?!]{1,2}\s`, "")
	rtkTabFile    = rx(`^\t\S`, "")
	rtkBuildErr   = rx(`\b(error|ERR!|fail(ed|ure)?|✖|✗|panic|exception|traceback|not ok|assert)`, "i")
	rtkBuildWarn  = rx(`\bwarn(ing)?\b`, "i")
	rtkProgress   = rx(`^\s*(Compiling|Downloading|Downloaded|Fetching|Installing|Resolving|Collecting|Building|Progress|\[\d+\/\d+\]|\d+%)`, "i")

func omitted(n int, what string) []uint16 { return cat(u16("… "), itoa16(n), u16(" "+what+" omitted")) }

func omittedIn(n int, what []uint16) []uint16 {
	return cat(u16("  … "), itoa16(n), u16(" "), what, u16(" omitted"))
}

func headTail(ls [][]uint16, head, tail int) [][]uint16 {
	if len(ls) <= head+tail+1 {
		return ls
	}
	out := make([][]uint16, 0, head+tail+1)
	out = append(out, ls[:head]...)
	out = append(out, omitted(len(ls)-head-tail, "lines"))
	return append(out, ls[len(ls)-tail:]...)
}

// dedupe: runs of lines alike once digits and hex ids are ignored collapse to one, then a blank line after a blank
// line goes (also a leading one: rtk.js reads a[-1] as the empty string).
func dedupe(ls [][]uint16, b *jsre.Budget) [][]uint16 {
	hash, zero := u16("#"), u16("0")
	toHash := func([]int) []uint16 { return hash }
	toZero := func([]int) []uint16 { return zero }
	// The keys are built in two scratch buffers reused for every line (prev keeps its own copy): a 10 MiB log would
	// otherwise allocate two keys per line (spec D19).
	var bufA, bufB []uint16
	keep := func(buf *[]uint16, k, src []uint16) []uint16 {
		if len(k) > 0 && (len(src) == 0 || &k[0] != &src[0]) {
			*buf = k[:0] // a replacement wrote a new buffer: reuse its storage next time
		}
		return k
	}
	key := func(l []uint16) []uint16 {
		k := keep(&bufA, rtkHexID.ReplaceAllInto(bufA, l, toHash, b), l)
		return trim16(keep(&bufB, rtkDigits.ReplaceAllInto(bufB, k, toZero, b), k))
	}
	var out [][]uint16
	var prev []uint16
	have, run := false, 0
	flush := func() {
		if run > 1 {
			out = append(out, cat(u16("  (× "), itoa16(run), u16(" similar lines)")))
		}
	}
	for _, l := range ls {
		b.Charge(1)
		k := key(l)
		if have && len(k) > 0 && equal16(k, prev) {
			run++
			continue
		}
		flush()
		out = append(out, l)
		prev, have, run = append(prev[:0], k...), true, 1
	}
	flush()
	res := make([][]uint16, 0, len(out))
	for i, l := range out {
		if blank(l) && (i == 0 || blank(out[i-1])) {
			continue
		}
		res = append(res, l)
	}
	return res
}

func filterGitDiff(t []uint16, b *jsre.Budget) []uint16 {
	var out [][]uint16
	hunk := 0
	for _, l := range splitNL(t) {
		b.Charge(1)
		if rtkDiffHead.TestB(l, b) || hasPrefix16(l, "@@") {
			out = append(out, l)
			hunk = 0
			continue
		}
		hunk++
		if hunk <= 60 {
			out = append(out, l)
		} else if hunk == 61 {
			out = append(out, u16("  … rest of hunk omitted"))
		}
	}
	return joinNL(headTail(out, 300, 100))
}

func filterGitLog(t []uint16, b *jsre.Budget) []uint16 {
	var kept [][]uint16
	for _, l := range splitNL(t) {
		b.Charge(1)
		if rtkLogHead.TestB(l, b) || rtkLogMsg.TestB(l, b) {
			kept = append(kept, l)
		}
	}
	var out [][]uint16
	for i, l := range kept {
		if !rtkIndent4.TestB(l, b) || i == 0 || !rtkIndent4.TestB(kept[i-1], b) {
			out = append(out, l)
		}
	}
	return joinNL(headTail(out, 150, 20))
}

func filterGitStatus(t []uint16, b *jsre.Budget) []uint16 {
	var out [][]uint16
	n := 0
	for _, l := range splitNL(t) {
		b.Charge(1)
		if rtkStatusFile.TestB(l, b) || rtkTabFile.TestB(l, b) {
			n++
			if n <= 25 {
				out = append(out, l)
			} else if n == 26 {
				out = append(out, u16("… more files omitted"))
			}
		} else {
			n = 0
			out = append(out, l)
		}
	}
	return joinNL(out)
}

func filterBuild(t []uint16, b *jsre.Budget) []uint16 {
	ls := splitNL(t)
	keep := make([]bool, len(ls))
	warn, progress := 0, 0
	for i, l := range ls {
		b.Charge(1)
		switch {
		case rtkBuildErr.TestB(l, b):
			for k := i; k <= i+3 && k < len(ls); k++ {
				keep[k] = true
			}
		case rtkBuildWarn.TestB(l, b) && func() bool { w := warn; warn++; return w < 5 }():
			keep[i] = true
		case rtkProgress.TestB(l, b): // a warning past the fifth that looks like progress is counted (spec D18)
			progress++
		}
	}
	var body [][]uint16
	for i, l := range ls {
		if keep[i] {
			body = append(body, l)
		}
	}
	var out [][]uint16
	if progress > 0 {
		out = append(out, cat(u16("("), itoa16(progress), u16(" progress lines)")))
	}
	d := dedupe(body, b)
	out = append(out, d[:min(len(d), 300)]...)
	if len(body) > 0 {
		out = append(out, u16("…"))
	}
	out = append(out, ls[max(0, len(ls)-25):]...)
	return joinNL(out)
}

type group struct {
	name  []uint16
	lines [][]uint16 // the first `keep` lines: the filters print no more of a group (spec D19)
	n     int        // every line of the group
}

// groups keeps insertion order, as a JS Map does. keep is how many lines of a group the filter prints.
type groups struct {
	order []*group
	index map[string]*group
	keep  int
}

func (g *groups) add(name, line []uint16) {
	if g.index == nil {
		g.index = map[string]*group{}
	}
	k := key16(name)
	x := g.index[k]
	if x == nil {
		x = &group{name: name}
		g.index[k] = x
		g.order = append(g.order, x)
	}
	if x.n < g.keep {
		x.lines = append(x.lines, line)
	}
	x.n++
}

func filterGrep(t []uint16, b *jsre.Budget) []uint16 {
	by := groups{keep: 10}
	var other [][]uint16 // the first 20: the filter prints no more
	eachLine(t, func(l []uint16) {
		b.Charge(1)
		matched := false
		rtkGrepLine.EachB(l, b, func(m []int) { // ^ without m: at most one match, at 0
			if !matched {
				matched = true
				by.add(l[m[2]:m[3]], l)
			}
		})
		if !matched && !blank(l) && len(other) < 20 {
			other = append(other, l)
		}
	})
	var out [][]uint16
	for i, g := range by.order {
		if i+1 > 40 {
			out = append(out, omitted(len(by.order)-40, "more files"))
			break
		}
		out = append(out, g.lines...)
		if g.n > 10 {
			out = append(out, omittedIn(g.n-10, cat(u16("more matches in "), g.name)))
		}
	}
	out = append(out, other...)
	return joinNL(out)
}

func filterListing(t []uint16, b *jsre.Budget) []uint16 {
	by := groups{keep: 12}
	eachLine(t, func(l []uint16) {
		b.Charge(1)
		p := trim16(l)
		if len(p) == 0 || rtkNoiseDirs.TestB(p, b) {
			return
		}
		if m := rtkListPrefix.ExecB(p, 0, b); m != nil {
			p = cat(p[:m[0]], p[m[1]:])
		}
		dir := u16(".")
		if i := lastIndex16(p, '/'); i >= 0 {
			dir = p[:i]
		}
		by.add(dir, l)
	})
	var out [][]uint16
	for i, g := range by.order {
		if i+1 > 40 {
			out = append(out, omitted(len(by.order)-40, "more folders"))
			break
		}
		out = append(out, g.lines...)
		if g.n > 12 {
			out = append(out, omittedIn(g.n-12, cat(u16("more in "), g.name)))
		}
	}
	return joinNL(out)
}

func filterRead(t []uint16, _ *jsre.Budget) []uint16 { return joinNL(headTail(splitNL(t), 160, 60)) }

func filterLog(t []uint16, b *jsre.Budget) []uint16 {
	return joinNL(headTail(dedupe(splitNL(t), b), 200, 80))
}

// jsonParse is JSON.parse as the saver needs it: the tree; or syntax (JSON.parse throws); or unsupported (JSON.parse
// accepts it but engine.Parse cannot hold it: an escaped lone surrogate, nesting over MaxDepth). engine.Parse stops at
// the first such thing, which may come before a syntax error JS throws on (review m1, m2), so an unsupported text is
// checked with encoding/json's validator. Valid holds lone surrogate escapes and caps nesting at 10000; with more
// brackets than that the answer is unknown and the text stays unsupported (the caller hands off: exact either way).
func jsonParse(text []byte) (v *Value, syntax, unsupported bool) {
	v, err := Parse(text)
	switch {
	case err == nil:
		return v, false, false
	case !errors.Is(err, ErrUnsupported):
		return nil, true, false
	case json.Valid(text):
		return nil, false, true
	case bytes.Count(text, []byte("["))+bytes.Count(text, []byte("{")) <= 10000:
		return nil, true, false // not JSON: JSON.parse throws too
	}
	return nil, false, true
}

// pickFilter is rtk.js pickFilter. For json it also returns the parsed tree (rtkJSON writes from it). why is a hand-off
// reason: JSON.parse accepts the text but the engine cannot hold it (spec D6).
func pickFilter(text, hint []uint16, b *jsre.Budget) (string, *Value, string) {
	h := hintLower(hint)
	head := text[:min(len(text), 2048)]
	switch {
	case hintGitDiff.TestB(h, b) || headGitDiff.TestB(head, b):
		return "git-diff", nil, ""
	case hintGitLog.TestB(h, b) || headGitLog.TestB(head, b):
		return "git-log", nil, ""
	case hintGitStatus.TestB(h, b) || headGitStatus.TestB(head, b):
		return "git-status", nil, ""
	case hintBuild.TestB(h, b) || headBuild.TestB(head, b):
		return "build", nil, ""
	case hintGrep.TestB(h, b) || hintGrepName.TestB(hint, b) || len(headGrepLines.FindAllB(head, b)) >= 3:
		return "grep", nil, ""
	}
	if headJSON.TestB(head, b) && tailJSON.TestB(text[max(0, len(text)-64):], b) {
		b.Charge(len(text))
		v, syntax, unsupported := jsonParse([]byte(jsre.Str(text)))
		switch {
		case unsupported:
			return "", nil, saverJSON
		case !syntax:
			return "json", v, ""
		} // a syntax error: JSON.parse throws, not json
	}
	switch {
	case hintListing.TestB(h, b) || hintGlobName.TestB(hint, b):
		return "listing", nil, ""
	case hintRead.TestB(hint, b) || len(headReadLines.FindAllB(head, b)) >= 10:
		return "read", nil, ""
	case len(text) > 4000:
		return "log", nil, ""
	}
	return "", nil, ""
}

// compressText is rtk.js compressText on one text. The JS try/catch around a filter has no counterpart: no Go filter
// fails on a text it is given (spec D2).
func compressText(raw, hint []uint16, b *jsre.Budget) ([]uint16, *rtkHit, string) {
	if len(raw) < rtkMinChars || len(raw) > rtkMaxChars || hasPrefix16(raw, "[error]") {
		return nil, nil, ""
	}
	clean := rtkANSI.ReplaceAllB(raw, func([]int) []uint16 { return nil }, b)
	name, parsed, why := pickFilter(clean, hint, b)
	if why != "" {
		return nil, nil, why
	}
	text := clean
	switch {
	case name == "json":
		text = rtkJSON(parsed)
	case name != "":
		text = rtkFilters[name](clean, b)
	}
	if len(text) >= len(raw) || blank(text) {
		if len(clean) < len(raw) {
			return clean, &rtkHit{"ansi", len(raw) - len(clean)}, ""
		}
		return nil, nil, ""
	}
	if name == "" {
		name = "ansi"
	}
	return text, &rtkHit{name, len(raw) - len(text)}, ""
}

package engine

// The token saver's json filter: JSON.stringify(cut(JSON.parse(t))) in one pass over engine.Parse's tree (spec D6).
// Arrays keep their first 20 items and a "… N more items" string; objects below depth 12 are rebuilt (keys in JS
// order); deeper ones are written whole; a string over 2000 UTF-16 units becomes its first 2000 and "… (N chars)".

import (
	"strconv"

	"github.com/rifqifitriady/ZAAFPulse-deno-go/gw/guard/jsre"
)

func rtkJSON(v *Value) []uint16 { return u16(string(cutWrite(nil, v, 0))) }

func cutWrite(b []byte, v *Value, depth int) []byte {
	switch v.Kind {
	case Array:
		b = append(b, '[')
		n := len(v.A)
		for i := 0; i < min(n, 20); i++ {
			if i > 0 {
				b = append(b, ',')
			}
			b = cutWrite(b, v.A[i], depth+1)
		}
		if n > 20 {
			b = append(b, ',')
			b = QuoteJS(b, "… "+strconv.Itoa(n-20)+" more items")
		}
		return append(b, ']')
	case Object:
		if depth >= 12 {
			return append(b, v.Stringify()...)
		}
		b = append(b, '{')
		for i, k := range v.Keys() {
			if i > 0 {
				b = append(b, ',')
			}
			b = QuoteJS(b, k)
			b = append(b, ':')
			b = cutWrite(b, v.Get(k), depth+1)
		}
		return append(b, '}')
	case String:
		if units(v.S) > 2000 {
			return cutString(b, v.S)
		}
	}
	return append(b, v.Stringify()...)
}

// cutString writes `${s.slice(0, 2000)}… (${s.length} chars)` as JSON.stringify writes it: a surrogate pair cut in
// half leaves a lone high surrogate, written \udXXX in lower-case hex.
func cutString(b []byte, s string) []byte {
	us := u16(s)
	head := us[:2000]
	var lone uint16
	if c := head[1999]; c >= 0xd800 && c <= 0xdbff {
		lone, head = c, head[:1999]
	}
	b = QuoteJS(b, jsre.Str(head))
	b = b[:len(b)-1] // reopen the string
	if lone != 0 {
		const hexd = "0123456789abcdef"
		b = append(b, '\\', 'u', hexd[lone>>12], hexd[lone>>8&15], hexd[lone>>4&15], hexd[lone&15])
	}
	tail := QuoteJS(nil, "… ("+strconv.Itoa(len(us))+" chars)")
	return append(b, tail[1:]...)
}

package engine

// The token saver's walks (lib/gateway/rtk.js saveTokens; spec D3, D7, D8). scanSaver is the cheap part, run while
// planning: it builds the hints, finds every body on which the JS throws (the engine hands those off: JS answers),
// and lists the texts the saver will look at, as pointers into the plan's pending body. run shortens them in place.

import (
	"strings"
	"time"

	"github.com/rifqifitriady/ZAAFPulse-deno-go/gw/guard/jsre"
)

const (
	saverThrows     = "token saver: a message it cannot read"
	saverArgs       = "token saver: a tool call the engine cannot read"
	saverOverBudget = "token saver: over its step budget"
)

// saverTarget is one text the saver looks at: holder[key] is a string; hint is its tool call's "<name> <arg>".
type saverTarget struct {
	holder *Value
	key    string
	hint   []uint16
}

type saverJob struct{ targets []saverTarget }

// candidate: compressText can change it (a string of 600 to 10 MiB units that does not start with [error]).
func candidate(v *Value) bool {
	if v == nil || v.Kind != String || strings.HasPrefix(v.S, "[error]") {
		return false
	}
	n := units(v.S)
	return n >= rtkMinChars && n <= rtkMaxChars
}

// firstTruthy is `a.k1 || a.k2 || …` on a JSON value (a primitive or an array has none of these properties); nil when
// every one is falsy.
func firstTruthy(a *Value, keys ...string) *Value {
	for _, k := range keys {
		if v := a.Get(k); v.Truthy() {
			return v
		}
	}
	return nil
}

// hintOf is `${name} ${val || ''}`; ok is false when either ToString throws.
func hintOf(name, val *Value) ([]uint16, bool) {
	n, ok := jsToString(name)
	if !ok {
		return nil, false
	}
	rest := []uint16{}
	if val != nil {
		if rest, ok = jsToString(val); !ok {
			return nil, false
		}
	}
	return cat(n, u16(" "), rest), true
}

// scanSaver walks the body as saveTokens does. A nil job means nothing can change (JS then returns null). why is a
// hand-off reason.
func scanSaver(body *Value, format string) (*saverJob, string) {
	msgs := body.Get("messages")
	if msgs == nil || msgs.Kind != Array {
		return nil, ""
	}
	j := &saverJob{}
	hints := map[string][]uint16{}
	lookup := func(id *Value) []uint16 {
		if k, ok := mapKey(id); ok {
			return hints[k]
		}
		return nil
	}
	remember := func(id *Value, h []uint16) {
		if k, ok := mapKey(id); ok {
			hints[k] = h
		}
	}
	// texts: content as a string, or the text of each {type: "text"} item; a null item throws (c.type).
	texts := func(holder *Value, hint []uint16) bool {
		c := holder.Get("content")
		switch {
		case c != nil && c.Kind == String:
			if candidate(c) {
				j.targets = append(j.targets, saverTarget{holder, "content", hint})
			}
		case c != nil && c.Kind == Array:
			for _, x := range c.A {
				if x == nil || x.Kind == Null {
					return false
				}
				if ty := x.Get("type"); ty != nil && ty.Kind == String && ty.S == "text" && candidate(x.Get("text")) {
					j.targets = append(j.targets, saverTarget{x, "text", hint})
				}
			}
		}
		return true
	}
	for _, m := range msgs.A {
		if m == nil || m.Kind == Null {
			return nil, saverThrows
		}
		if format == "anthropic" {
			c := m.Get("content")
			if c == nil || c.Kind != Array {
				continue
			}
			for _, bl := range c.A {
				if bl == nil || bl.Kind == Null {
					return nil, saverThrows
				}
				ty := bl.Get("type")
				isType := func(s string) bool { return ty != nil && ty.Kind == String && ty.S == s }
				if isType("tool_use") {
					var val *Value
					if in := bl.Get("input"); in.Truthy() {
						val = firstTruthy(in, "command", "pattern", "file_path", "path")
					}
					h, ok := hintOf(bl.Get("name"), val)
					if !ok {
						return nil, saverThrows
					}
					remember(bl.Get("id"), h)
				}
				if !isType("tool_result") || bl.Get("is_error").Truthy() {
					continue
				}
				if !texts(bl, lookup(bl.Get("tool_use_id"))) {
					return nil, saverThrows
				}
			}
			continue
		}
		// OpenAI: for (const tc of m.tool_calls || []). A truthy non-array throws: a non-empty string iterates
		// characters, which have no .function; anything else is not iterable.
		if tcs := m.Get("tool_calls"); tcs.Truthy() {
			if tcs.Kind != Array {
				return nil, saverThrows
			}
			for _, tc := range tcs.A {
				if tc == nil || tc.Kind != Object {
					return nil, saverThrows // tc.function.arguments on null, a primitive or an array
				}
				f := tc.Get("function")
				if f == nil || f.Kind == Null {
					return nil, saverThrows
				}
				a := NewObject() // let a = {}; a parse error leaves it
				if text, parse := argsText(f.Get("arguments")); parse {
					v, syntax, unsupported := jsonParse([]byte(text))
					switch {
					case unsupported:
						return nil, saverArgs
					case !syntax:
						a = v
					}
				}
				if a.Kind == Null {
					return nil, saverThrows // a.command outside the try
				}
				h, ok := hintOf(f.Get("name"), firstTruthy(a, "command", "cmd", "pattern", "file_path", "path", "input"))
				if !ok {
					return nil, saverThrows
				}
				remember(tc.Get("id"), h)
			}
		}
		if r := m.Get("role"); r == nil || r.Kind != String || r.S != "tool" {
			continue
		}
		if !texts(m, lookup(m.Get("tool_call_id"))) {
			return nil, saverThrows
		}
	}
	if len(j.targets) == 0 {
		return nil, ""
	}
	return j, ""
}

// argsText is the text JSON.parse(tc.function.arguments || '{}') reads. parse is false when String(arguments) throws
// inside the try (an object with an own toString key): a stays {}.
func argsText(args *Value) (string, bool) {
	switch {
	case !args.Truthy():
		return "{}", true
	case args.Kind == String:
		return args.S, true
	}
	s, ok := jsToString(args)
	if !ok {
		return "", false
	}
	return jsre.Str(s), true
}

// budget is spec D11's limit for this job.
func (j *saverJob) budget(now time.Time) *jsre.Budget {
	var n int64
	for _, t := range j.targets {
		n += int64(units(t.holder.Get(t.key).S))
	}
	return &jsre.Budget{Limit: 64*n + 1<<24, Deadline: now.Add(2 * time.Second)}
}

// run shortens every target in place. A hand-off reason leaves the body half done: the plan is dropped then.
func (j *saverJob) run(b *jsre.Budget) (hits []rtkHit, why string) {
	work := func() {
		for _, t := range j.targets {
			text, hit, w := compressText(u16(t.holder.Get(t.key).S), t.hint, b)
			if w != "" {
				why = w
				return
			}
			if hit != nil {
				t.holder.Set(t.key, Str(jsre.Str(text)))
				hits = append(hits, *hit)
			}
		}
	}
	if b == nil {
		work()
		return hits, why
	}
	if err := b.Run(work); err != nil {
		return nil, saverOverBudget
	}
	return hits, why
}

// SaveResult is saveTokens's result: savedTokens is Math.round(savedChars / 4).
type SaveResult struct {
	SavedChars, SavedTokens int
	Hits                    []rtkHit
}

// SaveTokens is saveTokens(body, format) in place: nil when nothing was shortened.
func SaveTokens(body *Value, format string, b *jsre.Budget) (*SaveResult, string) {
	j, why := scanSaver(body, format)
	if why != "" || j == nil {
		return nil, why
	}
	hits, why := j.run(b)
	if why != "" || len(hits) == 0 {
		return nil, why
	}
	chars := 0
	for _, h := range hits {
		chars += h.Saved
	}
	return &SaveResult{SavedChars: chars, SavedTokens: (chars + 2) / 4, Hits: hits}, ""
}

// Save is saveTokens on the plan's pending body (spec D8-D12): it shortens the texts scanSaver found, puts savedTokens
// in the plan's base line, and builds the provider body unless a guardrail check will (Guard builds it from the same
// pending body). It runs once per plan. why is a hand-off reason: the control plane answers the request.
func (s *State) Save(p *Plan, now time.Time) string {
	if p.saver == nil || p.saverDone {
		return ""
	}
	p.saverDone = true
	hits, why := p.saver.run(p.saver.budget(now))
	if why != "" {
		return why
	}
	if len(hits) > 0 {
		chars := 0
		for _, h := range hits {
			chars += h.Saved
		}
		n := int64((chars + 2) / 4)
		p.Base.SavedTokens = &n
		p.saverHits = len(hits)
	}
	if p.Policy == nil {
		p.Body = finishBody(p.Target, p.pending) // after P4d: p.build(p, p.pending)
	}
	return ""
}

// provider shortened, the usage line carries savedTokens, and the header turns it off. Under ZP_ENGINE=go the go
// engine serves these calls (the line carries "ipc"); under ZP_ENGINE=shadow every compare is the same.
const test = require('node:test');
const assert = require('node:assert/strict');
const { start } = require('./harness');

const GO = process.env.ZP_ENGINE === 'go';
let h;
test.before(async () => {
  h = await start();
});
test.after(async () => {
  if (h) await h.stop();
});

const DIFF = Array.from({ length: 3 }, (_, f) =>
  [`diff --git a/f${f}.js b/f${f}.js`, '@@ -1,90 +1,90 @@', ...Array.from({ length: 90 }, (_, i) => `+ line ${i}`)].join('\n'),
).join('\n');
const chatBody = (extra = {}) => ({
  model: 'oa/m1',
  messages: [
    { role: 'user', content: 'go' },
    { role: 'assistant', content: null, tool_calls: [{ id: 'c1', type: 'function', function: { name: 'Bash', arguments: '{"command":"git diff"}' } }] },
    { role: 'tool', tool_call_id: 'c1', content: DIFF },
  ],
  ...extra,
});
async function lineAfter(before) {
  const end = Date.now() + 5000;
  for (;;) {
    const logs = await h.logs();
    if (logs.length > before) return logs[0];
    if (Date.now() > end) throw new Error('no usage line within 5 s');
    await new Promise((r) => setTimeout(r, 20));
  }
}
const lastUp = (p) => h.upstream.requests.filter((x) => x.path === p).at(-1);

test('chat: the tool result reaches the provider shortened, savedTokens on the line', async () => {
  const before = (await h.logs()).length;

  assert.equal(r.status, 200);
  const sent = lastUp('/chat/completions').body.messages[2].content;
  assert.ok(sent.length < DIFF.length && sent.includes('… rest of hunk omitted'));
  const line = await lineAfter(before);

});

  assert.equal(r.status, 200);

  const line = await lineAfter(before);
  assert.ok(line.savedTokens > 0);
  if (GO) assert.ok(line.ipc);
});

test('x-zp-token-saver: off sends the text unchanged and writes no savedTokens', async () => {
  const before = (await h.logs()).length;
  await h.call('/v1/chat/completions', { key: h.key, body: chatBody(), headers: { 'x-zp-token-saver': 'off' } });
  assert.equal(lastUp('/chat/completions').body.messages[2].content, DIFF);

});

  assert.equal(h.upstream.requests.length, n);
});

test('/v1/messages with a tool_result: shortened on the translated body', async () => {

  assert.ok(tool.content.length < DIFF.length);

});
