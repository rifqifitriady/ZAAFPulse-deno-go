# P4e Guardrails and Budget Guard Wiring Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Go engine serves OpenAI-format chat calls (whole and streamed) that a guardrail check or a blocking budget applies to, byte-identical to the JS gateway: redaction, `log` hits, 400 `guardrail_blocked`, 500 `guardrails failed`, 402 budget and price refusals; budgets fail open with every unchecked call marked and alerted, and the request path never waits for a hung control plane (slices S6 and S7).

**Architecture:** `PlanChat` stays pure and cheap: it records the policy and keeps the upstream body. `Serve` then runs, in the JS order, the rate limit, `State.Budget` (price rule, `ipc.Decide` on the newest snapshot, the engine's own spend since that snapshot) and `State.Guard` (`gw/guard.CheckBody` on the body converted in memory to `jsv`), and only then calls the provider. Usage lines leave through the outbox and one sender goroutine, so a hung control plane never blocks a call. The shadow compares guarded calls and refusals off the forwarding path. Fixtures come from the real JS (`scripts/fixtures/gen-chat-guard.js`).

**Tech Stack:** Go 1.27.3 standard library only (`GOTOOLCHAIN=local`), Deno 2.9.8 and Node 24, CommonJS, `node:test`, IPC v1 (no message change), GitHub Actions, prettier 3 (printWidth 120) on clean JS files, `gofmt`.

```go
func hungGateway(t *testing.T, provider string, snapshot string, extra ...string) (*gwRun, int) {
	t.Helper()
	cp := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		_, _ = io.WriteString(w, "control plane")
	}))
	t.Cleanup(cp.Close)
	port := freePort(t)
	r := drive(t, extra...)
	r.next(t, "hello")
	r.send(`{"v":1,"type":"hello.ok","seq":1,"ts":0,"re":"h1","body":{"protocol":1,"cpVersion":"test"}}`)

	r.send(cfg)
	r.send(secretsLine(3, 1))
	r.send(snapshot)
	r.next(t, "ready")
	return r, port
}
```

```go
// SeqOf is the seq of an encoded event line.
func SeqOf(line []byte) (int64, error) { return seqOf(line) }

// Replay calls send with every stored line whose seq is at least from, in order, under the stream's send lock: no
// new event is stored or sent until the replay is done. The state lock is not held across a send, so an ack is
// never kept waiting by a slow reader (P3 Task 4 minor).
func (o *Outbox) Replay(from int64, send func(line []byte) error) error {
	o.sendMu.Lock()
	defer o.sendMu.Unlock()
	return o.scan(from, func(_ int64, line []byte) error { return send(line) })
}

// ReadFrom calls fn with every stored line whose seq is at least from, in order, without the send lock: events may be
// appended while it reads. A segment is append-only and a line not yet ended by a newline is not read, so fn sees
// whole lines only; a segment an ack pruned meanwhile is skipped (every event in it is stored by the control plane).
// The gateway's sender reads with it after its queue overflowed (P4e Decision 19).
func (o *Outbox) ReadFrom(from int64, fn func(seq int64, line []byte) error) error {
	return o.scan(from, fn)
}

func (o *Outbox) scan(from int64, fn func(seq int64, line []byte) error) error {
	o.mu.Lock()
	if o.closed {
		o.mu.Unlock()
		return ErrOutboxClosed
	}
	segs := slices.Clone(o.segs)
	o.mu.Unlock()
	for i, first := range segs {
		if i+1 < len(segs) && segs[i+1] <= from {
			continue
		}
		data, err := os.ReadFile(o.segPath(first))
		if errors.Is(err, fs.ErrNotExist) {
			continue
		}
		if err != nil {
			return err
		}
		for len(data) > 0 {
			nl := bytes.IndexByte(data, '\n')
			if nl < 0 {
				break
			}
			line := data[:nl+1]
			data = data[nl+1:]
			seq, err := seqOf(line[:nl])
			if err != nil {
				return err
			}
			if seq < from {
				continue
			}
			if err := fn(seq, line); err != nil {
				return err
			}
		}
	}
	return nil
}
```

```go
package main

import (
	"errors"
	"sync"
	"time"

	"github.com/rifqifitriady/ZAAFPulse-deno-go/gw/ipc"
)

// senderQueue is how many stored lines the sender holds in memory before it reads the rest from the outbox.
const senderQueue = 4096

// sender writes the outbox's events to the control plane from one goroutine, in seq order, so that no caller waits for
// the control plane to read (P4e Decision 19). The outbox's AppendSend still stores each event and hands it over under
// one lock (D4, E10), with enqueue as its send: lines arrive here in seq order and enqueue never blocks. When the queue
// is full the line is dropped from memory only (it is stored), the sender drains what it holds, then reads the rest
// from the outbox. A send error ends the sender: the control plane is gone, and the next start replays the outbox.
type sender struct {
	box  *ipc.Outbox
	send func(line []byte) error
	logf func(string, ...any)

	queue chan []byte
	wake  chan struct{}
	quit  chan struct{}
	done  chan struct{}
	once  sync.Once

	mu      sync.Mutex
	lost    bool // a line did not fit the queue: the rest comes from the outbox
	started bool

	next int64 // the seq the control plane gets next (the goroutine only)
}

func newSender(box *ipc.Outbox, send func([]byte) error, logf func(string, ...any)) *sender {
	return &sender{box: box, send: send, logf: logf, queue: make(chan []byte, senderQueue),
		wake: make(chan struct{}, 1), quit: make(chan struct{}), done: make(chan struct{})}
}

// start begins sending at seq next (after the handshake's replay sent everything before it).
func (s *sender) start(next int64) {
	s.mu.Lock()
	s.started = true
	s.mu.Unlock()
	s.next = next
	go s.run()
}

// enqueue is AppendSend's send: it never blocks.
func (s *sender) enqueue(line []byte) error {
	s.mu.Lock()
	if !s.lost {
		select {
		case s.queue <- line:
			s.mu.Unlock()
			return nil
		default:
			s.lost = true
		}
	}
	s.mu.Unlock()
	select {
	case s.wake <- struct{}{}:
	default:
	}
	return nil
}

func (s *sender) run() {
	defer close(s.done)
	for {
		select {
		case line := <-s.queue:
			if !s.one(line) {
				return
			}
		case <-s.wake:
		case <-s.quit:
			s.flush()
			return
		}
		if !s.catchUp() {
			return
		}
	}
}

// one sends a queued line, unless the outbox read already sent it; a line past the next seq means one was lost.
func (s *sender) one(line []byte) bool {
	seq, err := ipc.SeqOf(line)
	if err != nil || seq < s.next {
		return true
	}
	if seq > s.next {
		s.mu.Lock()
		s.lost = true
		s.mu.Unlock()
		return true
	}
	if err := s.send(line); err != nil {
		s.logf("ipc: sending to the control plane failed: %v (the outbox keeps the events)", err)
		return false
	}
	s.next = seq + 1
	return true
}

// catchUp reads from the outbox what the queue dropped, once the queue is empty.
func (s *sender) catchUp() bool {
	s.mu.Lock()
	if !s.lost || len(s.queue) > 0 {
		s.mu.Unlock()
		return true
	}
	s.lost = false // before the read: a line dropped from now on sets it again
	s.mu.Unlock()
	err := s.box.ReadFrom(s.next, func(seq int64, line []byte) error {
		if seq < s.next {
			return nil
		}
		if err := s.send(line); err != nil {
			return err
		}
		s.next = seq + 1
		return nil
	})
	if errors.Is(err, ipc.ErrOutboxClosed) {
		return false
	}
	if err != nil {
		s.logf("ipc: sending stored events failed: %v (the outbox keeps them)", err)
		return false
	}
	return true
}

// flush sends what is queued (and what the queue dropped) before stopping.
func (s *sender) flush() {
	for {
		select {
		case line := <-s.queue:
			if !s.one(line) {
				return
			}
		default:
			s.catchUp()
			return
		}
	}
}

// stop sends what it holds for at most grace, then gives up (the outbox keeps the rest for the next start). Call it
// before the outbox closes.
func (s *sender) stop(grace time.Duration) {
	s.mu.Lock()
	started := s.started
	s.mu.Unlock()
	if !started {
		return
	}
	s.once.Do(func() { close(s.quit) })
	select {
	case <-s.done:
	case <-time.After(grace):
		s.logf("ipc: %s passed with events still unsent; the next start sends them", grace)
	}
}
```

```go
// from the global setting; the scope is global only (the key's is ignored). The global policy was cleaned by
// lib/front/push.js globalPolicy and the key's modes by keyAuth.
func PolicyFor(global ipc.GuardrailsPolicy, key *ipc.GatewayKeyAuth) guard.Policy {
	p := guard.Policy{Scope: global.Scope, Secrets: global.Secrets, PII: global.PII, Injection: global.Injection}
	if key == nil {
		return p
	}
	own := key.Guardrails
	if own.Secrets != nil {
		p.Secrets = *own.Secrets
	}
	if own.PII != nil {
		p.PII = *own.PII
	}
	if own.Injection != nil {
		p.Injection = *own.Injection
	}
	return p
}

// checksApply is inScope(policy, tier) with at least one check on. With every check off, checkBody would return the
// body unchanged and no outcome, so skipping it is the same answer.
func checksApply(p guard.Policy, tier string) bool {
	return (p.Scope == "all" || tier != "private") && (p.Secrets != "off" || p.PII != "off" || p.Injection != "off")
}

// finishBody is callChat's provider body: { ...body, model: t.model } through adaptChat, as JSON.stringify writes it.
func finishBody(t *Target, out *Value) []byte {
	out.Set("model", Str(t.Model))
	AdaptChat(t.Provider, t.Model, out)
	return out.Stringify()

// toJSV is the engine's value as the guard reads it. Strings are shared, not copied.
func toJSV(v *Value) jsv.Value {
	switch v.Kind {
	case Null:
		return nil
	case Bool:
		return v.B
	case Number:
		return v.N
	case String:
		return v.S
	case Array:
		a := &jsv.Array{Items: make([]jsv.Value, len(v.A))}
		for i, x := range v.A {
			a.Items[i] = toJSV(x)
		}
		return a
	}
	o := jsv.NewObject()
	for _, k := range v.Keys() {
		o.Set(k, toJSV(v.Get(k)))
	}
	return o
}

// fromJSV is the guard's checked value back as the engine's (the guard never makes undefined).
func fromJSV(x jsv.Value) *Value {
	switch t := x.(type) {
	case bool:
		return &Value{Kind: Bool, B: t}
	case float64:
		return Num(t)
	case string:
		return Str(t)
	case *jsv.Array:
		v := &Value{Kind: Array, A: make([]*Value, len(t.Items))}
		for i, it := range t.Items {
			v.A[i] = fromJSV(it)
		}
		return v
	case *jsv.Object:
		o := NewObject()
		for _, k := range t.Keys() {
			o.Set(k, fromJSV(t.Get(k)))
		}
		return o
	}
	return &Value{Kind: Null}
}

func outcome(action string, hits []guard.Hit, failed bool) *ipc.GuardrailOutcome {
	o := &ipc.GuardrailOutcome{Action: action, Hits: make([]ipc.GuardrailHit, 0, len(hits))}
	for _, h := range hits {
		o.Hits = append(o.Hits, ipc.GuardrailHit{Check: h.Check, Label: h.Label, Count: int64(h.Count)})
	}
	if failed {
		t := true
		o.Error = &t
	}
	return o
}

// errorBody is lib/gateway/index.js fail's body: { error: { message, type, code? } }.
func errorBody(msg, typ, code string) []byte {
	b, e := NewObject(), NewObject()
	e.Set("message", Str(msg))
	e.Set("type", Str(typ))
	if code != "" {
		e.Set("code", Str(code))
	}
	b.Set("error", e)
	return b.Stringify()
}

// refusal is guard()'s rec(...) then throw: the line proxyChat's rec writes (the call's base without routedTo, no
// tokens, cost 0) and the client's answer.
func refusal(p *Plan, status int, msg, typ, code string, ms float64, lineErr string, g *ipc.GuardrailOutcome) *Answer {
	rec := p.Base
	rec.RoutedTo, rec.Guard = nil, g
	rec.Status, rec.Ms, rec.Cost = int64(status), ms, 0
	rec.Input, rec.Output, rec.Cached = nil, nil, nil
	rec.Error = &lineErr
	return &Answer{Status: status, Header: answerHeader(), Body: errorBody(msg, typ, code), Usage: rec}
}

// Guard runs the checks of a plan that has them and finishes its provider body; it returns the client's answer when
// the call is refused (its Usage is the line to emit), nil when the call goes on. ms is the time since the start.
func (s *State) Guard(p *Plan, ms float64) *Answer {
	if p.Policy == nil || p.pending == nil {
		return nil
	}
	res, err := checkBody(toJSV(p.pending), *p.Policy, "openai-chat")
	if err != nil {
		p.pending = nil
		var b *guard.Block
		if errors.As(err, &b) {
			p.Overrun = b.Overrun
			return refusal(p, http.StatusBadRequest, b.Error(), "invalid_request_error", "guardrail_blocked", ms,
				"blocked by guardrails", outcome("blocked", b.Hits, false))
		}
		return refusal(p, http.StatusInternalServerError, "guardrails failed", "server_error", "", ms,
			"guardrails failed", nil)
	}
	body := p.pending
	if res.Action == "redacted" {
		body = fromJSV(res.Body)
	}
	p.pending = nil
	p.Body = finishBody(p.Target, body)
	p.Failures = res.Failures
	if res.Action != "" {
		p.Base.Guard = outcome(res.Action, res.Hits, res.Error)
	}
	return nil
}
```

```go
const maxSpent = 65536

type Spent struct {

	In  ipc.CheckInput
	USD float64

// Spend holds the engine's own lines newer than the newest snapshot.

// Add keeps one line.
func (sp *Spend) Add(x Spent) {
	sp.mu.Lock()
	defer sp.mu.Unlock()
	if len(sp.lines) >= maxSpent {
		sp.lines = append(sp.lines[:0], sp.lines[len(sp.lines)-maxSpent/2:]...)

	sp.lines = append(sp.lines, x)

func (sp *Spend) Since(at int64) []Spent {

	for _, l := range sp.lines {
		if l.Ts > at {
			keep = append(keep, l)
		}
	}
	sp.lines = keep
	return append([]Spent(nil), keep...)
```

```go
// budgetLag is the reason a budget disagreement is not a diff: the control plane decides on its live spend, the engine
// on its snapshot, up to a second behind (spec Decision 18).
const budgetLag = "budget: the go engine decided on its snapshot"

// CompareRefusal is the line for a request the engine would refuse itself (ref: a 402, or a guardrail 400/500).
func (s *State) CompareRefusal(class string, p *Plan, ref *Answer, c *Capture) ShadowLine {
	l := ShadowLine{Class: class}
	if c != nil && c.Client != nil {
		l.Status = c.Client.Status
	}
	d := func(field string, js, gov any) { l.Diffs = append(l.Diffs, Diff{field, js, gov}) }
	switch {
	case c == nil || c.Client == nil:
		d("served", "no capture", "refused "+NumberString(float64(ref.Status)))
	case len(c.ProviderRequests) > 0 || c.ProviderResponse != nil:
		if ref.Status == http.StatusPaymentRequired {
			l.Outcome, l.Reason = "skipped", budgetLag
			return l
		}
		d("served", "called the provider", "refused "+NumberString(float64(ref.Status)))
	default:
		if c.Client.Status != ref.Status {
			if c.Client.Status == http.StatusPaymentRequired || ref.Status == http.StatusPaymentRequired {
				l.Outcome, l.Reason = "skipped", budgetLag
				return l
			}
			d("client.status", c.Client.Status, ref.Status)
		}
		if c.Client.Body != "" && c.Client.Body != string(ref.Body) {
			a, b := digest(c.Client.Body, string(ref.Body))
			d("client.body", a, b)
		}
		switch {
		case ref.NoLine && c.Usage != nil:
			d("usage", "a line", "none")
		case !ref.NoLine && c.Usage == nil:
			d("usage", "none", "a line")
		case !ref.NoLine:
			var goU map[string]any
			b, _ := json.Marshal(ref.Usage)
			_ = json.Unmarshal(b, &goU)
			for _, f := range usageFields {
				if canonical(c.Usage[f]) != canonical(goU[f]) {
					d("usage."+f, c.Usage[f], goU[f])
				}
			}
			if js, gov := text(c.Usage["error"]), text(goU["error"]); js != gov {
				a, b := digest(js, gov)
				d("usage.error", a, b)
			}
		}
	}
	l.Outcome = "same"
	if len(l.Diffs) > 0 {
		l.Outcome = "diff"
	}
	return l
}
```
